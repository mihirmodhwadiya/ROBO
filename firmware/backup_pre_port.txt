void takeAllServoMutexes() {
    if (leftServoMutex != NULL) {
        xSemaphoreTake(leftServoMutex, portMAX_DELAY);
    }

    if (rightServoMutex != NULL) {
        xSemaphoreTake(rightServoMutex, portMAX_DELAY);
    }
}

void giveAllServoMutexes() {
    if (rightServoMutex != NULL) {
        xSemaphoreGive(rightServoMutex);
    }

    if (leftServoMutex != NULL) {
        xSemaphoreGive(leftServoMutex);
    }
}

int getServoMaxAngle(const String& name) {

    // Shoulders: maximum 90 degrees
    if (name == "R1_SHOUL_Servo" ||
        name == "R2_SHOUL_Servo" ||
        name == "L1_SHOUL_Servo" ||
        name == "L2_SHOUL_Servo") {

        return SHOULDER_MAX_ANGLE;
    }

    // Elbows: maximum 180 degrees
    if (name == "R3_EL_Servo" ||
        name == "L4_EL_Servo" ||
        name == "L3_EL_Servo" ||
        name == "R4_EL_Servo") {

        return ELBOW_MAX_ANGLE;
    }

    return -1;
}

// =====================================================
// DEFAULT ANGLE LOOKUP
// =====================================================

int getServoDefaultAngle(const String& name) {

    if (name == "R1_SHOUL_Servo") {
        return DEFAULT_R1_SHOULDER;
    }

    if (name == "R2_SHOUL_Servo") {
        return DEFAULT_R2_SHOULDER;
    }

    if (name == "L1_SHOUL_Servo") {
        return DEFAULT_L1_SHOULDER;
    }

    if (name == "L2_SHOUL_Servo") {
        return DEFAULT_L2_SHOULDER;
    }

    if (name == "R3_EL_Servo") {
        return DEFAULT_R3_ELBOW;
    }

    if (name == "L4_EL_Servo") {
        return DEFAULT_L4_ELBOW;
    }

    if (name == "L3_EL_Servo") {
        return DEFAULT_L3_ELBOW;
    }

    if (name == "R4_EL_Servo") {
        return DEFAULT_R4_ELBOW;
    }

    return -1;
}

// =====================================================
// STRICT SERVO CONTROL
//
// Valid commands:
//   Shoulders: 0-90
//   Elbows:    0-180
//
// If the requested angle is outside the permitted range,
// that servo is reset to its own configured default.
//
// Unknown servo names are rejected.
// =====================================================

bool setServo(const String& name, long requestedAngle) {

    int maxAngle = getServoMaxAngle(name);
    int defaultAngle = getServoDefaultAngle(name);

    if (maxAngle < 0 || defaultAngle < 0) {
        Serial.println("ERROR: Unknown servo: " + name);
        return false;
    }

    int targetAngle;

    if (requestedAngle < SHOULDER_MIN_ANGLE ||
        requestedAngle > maxAngle) {

        targetAngle = defaultAngle;

        Serial.println();
        Serial.println("==========================================");
        Serial.println("SERVO ANGLE LIMIT EXCEEDED");
        Serial.println("Servo: " + name);

        Serial.printf(
            "Requested angle: %ld degrees\n",
            requestedAngle
        );

        Serial.printf(
            "Allowed range: 0-%d degrees\n",
            maxAngle
        );

        Serial.printf(
            "Reset position: %d degrees\n",
            defaultAngle
        );

        Serial.println("==========================================");
    }
    else {
        targetAngle = (int)requestedAngle;
    }

    // Left-side servo group, including R2 shoulder.
    bool isLeftGroup =
        name.startsWith("L") ||
        name == "R2_SHOUL_Servo";

    SemaphoreHandle_t mutex =
        isLeftGroup ? leftServoMutex : rightServoMutex;

    if (mutex == NULL) {
        Serial.println("ERROR: Servo mutex unavailable.");
        return false;
    }

    if (xSemaphoreTake(mutex, pdMS_TO_TICKS(100)) != pdTRUE) {
        Serial.println("ERROR: Could not acquire servo mutex.");
        return false;
    }

    // LEFT GROUP
    if (isLeftGroup) {

        if (name == "L1_SHOUL_Servo") {
            L1_SHOUL_Servo.write(targetAngle);
            angleL1_SHOUL_Servo = targetAngle;
        }

        else if (name == "L2_SHOUL_Servo") {
            L2_SHOUL_Servo.write(targetAngle);
            angleL2_SHOUL_Servo = targetAngle;
        }

        else if (name == "L3_EL_Servo") {
            L3_EL_Servo.write(targetAngle);
            angleL3_EL_Servo = targetAngle;
        }

        else if (name == "L4_EL_Servo") {
            L4_EL_Servo.write(targetAngle);
            angleL4_EL_Servo = targetAngle;
        }

        else if (name == "R2_SHOUL_Servo") {
            R2_SHOUL_Servo.write(targetAngle);
            angleR2_SHOUL_Servo = targetAngle;
        }
    }

    // RIGHT GROUP
    else {

        if (name == "R1_SHOUL_Servo") {
            R1_SHOUL_Servo.write(targetAngle);
            angleR1_SHOUL_Servo = targetAngle;
        }

        else if (name == "R3_EL_Servo") {
            R3_EL_Servo.write(targetAngle);
            angleR3_EL_Servo = targetAngle;
        }

        else if (name == "R4_EL_Servo") {
            R4_EL_Servo.write(targetAngle);
            angleR4_EL_Servo = targetAngle;
        }
    }

    xSemaphoreGive(mutex);

    return true;
}

// =====================================================
// TAB SWITCHING
// =====================================================

function openTab(tabId, element) {

    document.querySelectorAll('.panel').forEach(panel => {
        panel.classList.remove('active');
    });

    document.querySelectorAll('.tab-btn').forEach(button => {
        button.classList.remove('active');
    });

    document.getElementById('panel-' + tabId).classList.add('active');

    if (element) {
        element.classList.add('active');
    }

    if (tabId === 'export') {
        updateExportCode();
    }
}

// =====================================================
// SERVO DEFAULT VALUES
// =====================================================

const servoDefaults = {
    R1_SHOUL_Servo: 90,
    R2_SHOUL_Servo: 0,
    L1_SHOUL_Servo: 0,
    L2_SHOUL_Servo: 90,

    R3_EL_Servo: 180,
    L4_EL_Servo: 0,
    L3_EL_Servo: 0,
    R4_EL_Servo: 180
};

function getMaxForServo(name) {

    if (name.endsWith('_SHOUL_Servo')) {
        return 90;
    }

    return 180;
}

// =====================================================
// SEND ANGLE TO ESP32
// =====================================================

const pendingRequests = {};

function sendServoAngle(name, angle) {

    if (pendingRequests[name]) {
        clearTimeout(pendingRequests[name]);
    }

    pendingRequests[name] = setTimeout(() => {

        fetch('/servo?name=' + encodeURIComponent(name) +
              '&angle=' + encodeURIComponent(angle))

        .then(response => {

            if (!response.ok) {
                throw new Error('Servo command failed');
            }

            return response.text();
        })

        .then(message => {

            if (message.startsWith('OUT_OF_RANGE_RESET_TO_DEFAULT')) {
                refreshAllAngles();
            }
        })

        .catch(error => {
            console.error('Servo command error:', error);
        });

    }, 25);
}

// =====================================================
// SLIDER CONTROL
// =====================================================

function onSliderMove(name) {

    const slider = document.getElementById(name);

    const maxAngle = getMaxForServo(name);

    let angle = parseInt(slider.value, 10);

    angle = Math.max(0, Math.min(maxAngle, angle));

    slider.value = angle;

    document.getElementById('disp_' + name).innerText =
        angle + '°';

    sendServoAngle(name, angle);
}

// =====================================================
// SET EXACT ANGLE
// =====================================================

function setExact(name, angle) {

    const slider = document.getElementById(name);

    if (!slider) {
        return;
    }

    const maxAngle = getMaxForServo(name);

    angle = Math.max(0, Math.min(maxAngle, angle));

    slider.value = angle;

    document.getElementById('disp_' + name).innerText =
        angle + '°';

    sendServoAngle(name, angle);
}

// =====================================================
// NUDGE BY A SMALL ANGLE
// =====================================================

function nudge(name, delta) {

    const slider = document.getElementById(name);

    if (!slider) {
        return;
    }

    const current = parseInt(slider.value, 10);

    setExact(name, current + delta);
}

// =====================================================
// SYNCHRONIZE UI WITH ESP32 ANGLES
// =====================================================

function syncAngles(data) {

    Object.keys(data).forEach(name => {

        const slider = document.getElementById(name);
        const display = document.getElementById('disp_' + name);

        if (slider) {
            slider.value = data[name];
        }

        if (display) {
            display.innerText = data[name] + '°';
        }
    });
}

function refreshAllAngles() {

    fetch('/angles')

    .then(response => response.json())

    .then(data => {
        syncAngles(data);
    })

    .catch(error => {
        console.error('Could not refresh servo angles:', error);
    });
}

// =====================================================
// SET ALL DEFAULTS
// =====================================================

function presetDefault() {

    const button = document.getElementById('defaultButton');
    const status = document.getElementById('statusMessage');

    button.disabled = true;

    status.innerText = 'Resetting all servos to default positions...';

    // Show expected defaults immediately.
    syncAngles(servoDefaults);

    fetch('/preset?mode=default')

    .then(response => {

        if (!response.ok) {
            throw new Error('Default reset failed');
        }

        return response.text();
    })

    .then(message => {

        status.innerText = message;

        return fetch('/angles');
    })

    .then(response => response.json())

    .then(data => {

        syncAngles(data);

        status.innerText =
            'All eight servos reset to calibrated defaults.';
    })

    .catch(error => {

        console.error('Default reset error:', error);

        status.innerText =
            'Reset failed. Check the ESP32 Wi-Fi connection.';

        refreshAllAngles();
    })

    .finally(() => {
        button.disabled = false;
    });
}

// =====================================================
// SET ALL SHOULDERS
// =====================================================

function setAllShoulders(angle) {

    angle = Math.max(0, Math.min(90, angle));

    const names = [
        'R1_SHOUL_Servo',
        'R2_SHOUL_Servo',
        'L1_SHOUL_Servo',
        'L2_SHOUL_Servo'
    ];

    names.forEach((name, index) => {

        const slider = document.getElementById(name);
        const display = document.getElementById('disp_' + name);

        if (slider) {
            slider.value = angle;
        }

        if (display) {
            display.innerText = angle + '°';
        }

        setTimeout(() => {
            sendServoAngle(name, angle);
        }, index * 35);
    });
}

// =====================================================
// HTTP SERVO HANDLER
//
// Out-of-range requests reset the servo to its default.
// =====================================================

void handleServo() {

    if (!server.hasArg("name") || !server.hasArg("angle")) {

        server.send(400, "text/plain", "Missing parameters");

        return;
    }

    String name = server.arg("name");
    String rawAngle = server.arg("angle");

    if (getServoMaxAngle(name) < 0) {

        server.send(400, "text/plain", "Unknown servo name");

        return;
    }

    // Parse the complete angle string.
    const char* start = rawAngle.c_str();

    char* end = NULL;

    errno = 0;

    long requestedAngle = strtol(start, &end, 10);

    if (start == end || *end != '\0' || errno == ERANGE) {

        server.send(400, "text/plain", "Invalid angle");

        return;
    }

    int maxAngle = getServoMaxAngle(name);

    bool wasOutOfRange =
        requestedAngle < 0 ||
        requestedAngle > maxAngle;

    if (!setServo(name, requestedAngle)) {

        server.send(
            503,
            "text/plain",
            "Servo busy or mutex unavailable"
        );

        return;
    }

    if (wasOutOfRange) {

        int defaultAngle = getServoDefaultAngle(name);

        Serial.printf(
            "%s requested %ld degrees. Reset to %d degrees.\n",
            name.c_str(),
            requestedAngle,
            defaultAngle
        );

        server.send(
            200,
            "text/plain",
            "OUT_OF_RANGE_RESET_TO_DEFAULT: " +
            String(defaultAngle)
        );
    }

    else {

        Serial.printf(
            "%s set to %ld degrees.\n",
            name.c_str(),
            requestedAngle
        );

        server.send(200, "text/plain", "OK");
    }
}

// HTTP WALK HANDLER
// =====================================================

void handleWalk() {

    if (!server.hasArg("state")) {

        server.send(400, "text/plain", "Missing state");

        return;
    }

    int state = server.arg("state").toInt();

    if (state == 1) {

        // Stop Hi and 360 before starting walking.
        wavingHi = false;
        spinning360 = false;
        walking = false;

        takeAllServoMutexes();

        walkPhase = 0;

        // Set calibrated shoulder walking stance.
        R1_SHOUL_Servo.write(WALK_POS_R1_SHOULDER);
        angleR1_SHOUL_Servo = WALK_POS_R1_SHOULDER;
        delay(20);

        R2_SHOUL_Servo.write(WALK_POS_R2_SHOULDER);
        angleR2_SHOUL_Servo = WALK_POS_R2_SHOULDER;
        delay(20);

        L1_SHOUL_Servo.write(WALK_POS_L1_SHOULDER);
        angleL1_SHOUL_Servo = WALK_POS_L1_SHOULDER;
        delay(20);

        L2_SHOUL_Servo.write(WALK_POS_L2_SHOULDER);
        angleL2_SHOUL_Servo = WALK_POS_L2_SHOULDER;

        giveAllServoMutexes();

        walking = true;

        Serial.println("Parallel walking started.");

        server.send(200, "text/plain", "Walking");
    }

    else {

        walking = false;

        takeAllServoMutexes();

        // Return elbows to the walking vertical stance.
        R3_EL_Servo.write(VERTICAL_LEG_VALUE_OF_R3_EL_Servo);
        angleR3_EL_Servo = VERTICAL_LEG_VALUE_OF_R3_EL_Servo;
        delay(25);

        L4_EL_Servo.write(VERTICAL_LEG_VALUE_OF_L4_EL_Servo);
        angleL4_EL_Servo = VERTICAL_LEG_VALUE_OF_L4_EL_Servo;
        delay(25);

        L3_EL_Servo.write(VERTICAL_LEG_VALUE_OF_L3_EL_Servo);
        angleL3_EL_Servo = VERTICAL_LEG_VALUE_OF_L3_EL_Servo;
        delay(25);

        R4_EL_Servo.write(VERTICAL_LEG_VALUE_OF_R4_EL_Servo);
        angleR4_EL_Servo = VERTICAL_LEG_VALUE_OF_R4_EL_Servo;

        giveAllServoMutexes();

        Serial.println("Walking stopped.");

        server.send(200, "text/plain", "Stopped");
    }
}

// =====================================================
// SAY HI GESTURE
// =====================================================

void startHi() {

    walking = false;
    spinning360 = false;

    if (!wavingHi) {

        if (leftServoMutex != NULL &&
            xSemaphoreTake(leftServoMutex, portMAX_DELAY) == pdTRUE) {

            prevL4Angle = angleL4_EL_Servo;
            prevL3Angle = angleL3_EL_Servo;
            prevL1Angle = angleL1_SHOUL_Servo;

            xSemaphoreGive(leftServoMutex);
        }
    }

    waveTargetCycles = -1;
    waveCycle = 0;
    waveStep = 0;

    lastWaveTime = 0;

    wavingHi = true;

    Serial.println("Say Hi gesture started.");
}

void sayHi(int cycles) {

    if (cycles <= 0) {
        startHi();
        return;
    }

    walking = false;

    if (!wavingHi) {

        if (leftServoMutex != NULL &&
            xSemaphoreTake(leftServoMutex, portMAX_DELAY) == pdTRUE) {

            prevL4Angle = angleL4_EL_Servo;
            prevL3Angle = angleL3_EL_Servo;
            prevL1Angle = angleL1_SHOUL_Servo;

            xSemaphoreGive(leftServoMutex);
        }
    }

    waveTargetCycles = cycles;
    waveCycle = 0;
    waveStep = 0;

    lastWaveTime = 0;

    wavingHi = true;
}

void stopHi() {

    takeAllServoMutexes();

    if (wavingHi) {

        wavingHi = false;

        L4_EL_Servo.write(prevL4Angle);
        angleL4_EL_Servo = prevL4Angle;
        delay(35);

        L1_SHOUL_Servo.write(prevL1Angle);
        angleL1_SHOUL_Servo = prevL1Angle;
        delay(35);

        L3_EL_Servo.write(prevL3Angle);
        angleL3_EL_Servo = prevL3Angle;

        Serial.println("Say Hi stopped. Previous stance restored.");
    }

    giveAllServoMutexes();
}

void updateHi() {

    unsigned long currentTime = millis();

    if (currentTime - lastWaveTime < HI_WAVE_DELAY) {
        return;
    }

    lastWaveTime = currentTime;

    // Step 0: L3 balance
    if (waveStep == 0) {

        L3_EL_Servo.write(HI_BALANCE_L3_ANGLE);
        angleL3_EL_Servo = HI_BALANCE_L3_ANGLE;

        waveStep = 1;
    }

    // Step 1: Raise L1 shoulder
    else if (waveStep == 1) {

        L1_SHOUL_Servo.write(HI_SHOULDER_L1_ANGLE);
        angleL1_SHOUL_Servo = HI_SHOULDER_L1_ANGLE;

        waveStep = 2;
    }

    // Step 2: Raise L4
    else if (waveStep == 2) {

        L4_EL_Servo.write(HI_WAVE_MAX_ANGLE);
        angleL4_EL_Servo = HI_WAVE_MAX_ANGLE;

        waveStep = 3;
    }

    // Step 3: Lower L4
    else if (waveStep == 3) {

        L4_EL_Servo.write(HI_WAVE_MIN_ANGLE);
        angleL4_EL_Servo = HI_WAVE_MIN_ANGLE;

        waveStep = 4;
    }

    // Step 4: Raise L4 again
    else if (waveStep == 4) {

        L4_EL_Servo.write(HI_WAVE_MAX_ANGLE);
        angleL4_EL_Servo = HI_WAVE_MAX_ANGLE;

        if (waveTargetCycles > 0) {

            waveCycle++;

            if (waveCycle >= waveTargetCycles) {
                waveStep = 5;
            }

            else {
                waveStep = 3;
            }
        }

        else {
            // Continuous waving
            waveStep = 3;
        }
    }

    // Step 5: Restore L4
    else if (waveStep == 5) {

        L4_EL_Servo.write(prevL4Angle);
        angleL4_EL_Servo = prevL4Angle;

        waveStep = 6;
    }

    // Step 6: Restore L1 shoulder
    else if (waveStep == 6) {

        L1_SHOUL_Servo.write(prevL1Angle);
        angleL1_SHOUL_Servo = prevL1Angle;

        waveStep = 7;
    }

    // Step 7: Restore L3
    else if (waveStep == 7) {

        L3_EL_Servo.write(prevL3Angle);
        angleL3_EL_Servo = prevL3Angle;

        wavingHi = false;

        Serial.println("Say Hi gesture completed.");
    }
}

// =====================================================
// HTTP HI HANDLER
// =====================================================

void handleHi() {

    int state = 1;

    if (server.hasArg("state")) {
        state = server.arg("state").toInt();
    }

    if (state == 0) {

        stopHi();

        server.send(200, "text/plain", "Say Hi Stopped");
    }

    else {

        if (server.hasArg("cycles")) {

            int cycles = server.arg("cycles").toInt();

            sayHi(cycles);

            server.send(
                200,
                "text/plain",
                "Say Hi started for " + String(cycles) + " cycles"
            );
        }

        else {

            startHi();

            server.send(200, "text/plain", "Waving Hi (Continuous)");
        }
    }
}

// =====================================================
// 360 SPIN GESTURE (L2 x23) — Horizontal Position
// =====================================================

void start360(int cycles) {

    walking = false;
    wavingHi = false;

    if (!spinning360) {

        takeAllServoMutexes();

        prevL2Angle = angleL2_SHOUL_Servo;

        // Set elbows to horizontal (flat) position first.
        R3_EL_Servo.write(HORIZONTAL_LEG_VALUE_OF_R3_EL_Servo);
        angleR3_EL_Servo = HORIZONTAL_LEG_VALUE_OF_R3_EL_Servo;
        delay(25);

        L4_EL_Servo.write(HORIZONTAL_LEG_VALUE_OF_L4_EL_Servo);
        angleL4_EL_Servo = HORIZONTAL_LEG_VALUE_OF_L4_EL_Servo;
        delay(25);

        L3_EL_Servo.write(HORIZONTAL_LEG_VALUE_OF_L3_EL_Servo);
        angleL3_EL_Servo = HORIZONTAL_LEG_VALUE_OF_L3_EL_Servo;
        delay(25);

        R4_EL_Servo.write(HORIZONTAL_LEG_VALUE_OF_R4_EL_Servo);
        angleR4_EL_Servo = HORIZONTAL_LEG_VALUE_OF_R4_EL_Servo;
        delay(25);

        // Start L2 shoulder at 90 (default).
        L2_SHOUL_Servo.write(90);
        angleL2_SHOUL_Servo = 90;

        giveAllServoMutexes();
    }

    spin360TargetCycles = (cycles > 0) ? cycles : SPIN360_DEFAULT_CYCLES;
    spin360Cycle = 0;
    spin360Step = 0;

    lastSpin360Time = 0;

    spinning360 = true;

    Serial.printf("360 Spin started for %d cycles (L2: 90->0, horizontal pose).\n", spin360TargetCycles);
}

void stop360() {

    takeAllServoMutexes();

    if (spinning360) {

        spinning360 = false;

        // Restore L2 shoulder to default.
        L2_SHOUL_Servo.write(DEFAULT_L2_SHOULDER);
        angleL2_SHOUL_Servo = DEFAULT_L2_SHOULDER;

        Serial.println("360 Spin stopped. L2 default angle restored.");
    }

    giveAllServoMutexes();
}

void update360() {

    unsigned long currentTime = millis();

    if (currentTime - lastSpin360Time < SPIN360_DELAY) {
        return;
    }

    lastSpin360Time = currentTime;

    // Step 0: Sweep L2 towards 0 (horizontal sweep out)
    if (spin360Step == 0) {

        L2_SHOUL_Servo.write(0);
        angleL2_SHOUL_Servo = 0;

        spin360Step = 1;
    }

    // Step 1: Return L2 towards 90 (horizontal sweep back)
    else if (spin360Step == 1) {

        L2_SHOUL_Servo.write(90);
        angleL2_SHOUL_Servo = 90;

        spin360Cycle++;

        if (spin360Cycle >= spin360TargetCycles) {

            spinning360 = false;
            spin360Step = 0;

            Serial.printf("360 Spin finished all %d cycles.\n", spin360TargetCycles);
        }

        else {
            spin360Step = 0;
        }
    }
}

// =====================================================
// HTTP 360 HANDLER
// =====================================================

void handle360() {

    int state = 1;

    if (server.hasArg("state")) {
        state = server.arg("state").toInt();
    }

    if (state == 0) {

        stop360();

        server.send(200, "text/plain", "360 Spin Stopped");
    }

    else {

        int cycles = SPIN360_DEFAULT_CYCLES;

        if (server.hasArg("cycles")) {
            cycles = server.arg("cycles").toInt();
        }

        start360(cycles);

        server.send(
            200,
            "text/plain",
            "360 Spin started (" + String(cycles) + " cycles)"
        );
    }
}

// =====================================================
// HTTP PRESET HANDLER
// =====================================================

void handlePreset() {

    if (!server.hasArg("mode")) {

        server.send(400, "text/plain", "Missing mode");

        return;
    }

    String mode = server.arg("mode");

    // Restore all eight calibrated default positions.
    if (mode == "default" || mode == "defaults") {

        presetDefault();

        server.send(
            200,
            "text/plain",
            "All eight servos reset to calibrated defaults."
        );

        return;
    }

    if (mode == "hi" ||
        mode == "say_hi" ||
        mode == "start_hi") {

        startHi();

        server.send(200, "text/plain", "Waving Hi (Continuous)");

        return;
    }

    if (mode == "stop_hi") {

        stopHi();

        server.send(200, "text/plain", "Say Hi Stopped");

        return;
    }

    if (mode == "360" ||
        mode == "spin360" ||
        mode == "start_360") {

        start360(SPIN360_DEFAULT_CYCLES);

        server.send(200, "text/plain", "360 Spin started (23 cycles)");

        return;
    }

    if (mode == "stop_360" || mode == "stop360") {

        stop360();

        server.send(200, "text/plain", "360 Spin Stopped");

        return;
    }

    // Other presets stop both autonomous movements.
    walking = false;
    wavingHi = false;
    spinning360 = false;

    takeAllServoMutexes();

    // -----------------------------------------------
    // HORIZONTAL
    // -----------------------------------------------

    if (mode == "horizontal") {

        R3_EL_Servo.write(HORIZONTAL_LEG_VALUE_OF_R3_EL_Servo);
        angleR3_EL_Servo = HORIZONTAL_LEG_VALUE_OF_R3_EL_Servo;
        delay(25);

        L4_EL_Servo.write(HORIZONTAL_LEG_VALUE_OF_L4_EL_Servo);
        angleL4_EL_Servo = HORIZONTAL_LEG_VALUE_OF_L4_EL_Servo;
        delay(25);

        L3_EL_Servo.write(HORIZONTAL_LEG_VALUE_OF_L3_EL_Servo);
        angleL3_EL_Servo = HORIZONTAL_LEG_VALUE_OF_L3_EL_Servo;
        delay(25);

        R4_EL_Servo.write(HORIZONTAL_LEG_VALUE_OF_R4_EL_Servo);
        angleR4_EL_Servo = HORIZONTAL_LEG_VALUE_OF_R4_EL_Servo;
    }

    // -----------------------------------------------
    // VERTICAL WALKING STANCE
    // -----------------------------------------------

    else if (mode == "vertical") {

        R3_EL_Servo.write(VERTICAL_LEG_VALUE_OF_R3_EL_Servo);
        angleR3_EL_Servo = VERTICAL_LEG_VALUE_OF_R3_EL_Servo;
        delay(25);

        L4_EL_Servo.write(VERTICAL_LEG_VALUE_OF_L4_EL_Servo);
        angleL4_EL_Servo = VERTICAL_LEG_VALUE_OF_L4_EL_Servo;
        delay(25);

        L3_EL_Servo.write(VERTICAL_LEG_VALUE_OF_L3_EL_Servo);
        angleL3_EL_Servo = VERTICAL_LEG_VALUE_OF_L3_EL_Servo;
        delay(25);

        R4_EL_Servo.write(VERTICAL_LEG_VALUE_OF_R4_EL_Servo);
        angleR4_EL_Servo = VERTICAL_LEG_VALUE_OF_R4_EL_Servo;
    }

    // -----------------------------------------------
    // ALL ELBOWS 45 DEGREES
    // -----------------------------------------------

    else if (mode == "elbow_45" || mode == "45") {

        R3_EL_Servo.write(DEG45_LEG_VALUE_OF_R3_EL_Servo);
        angleR3_EL_Servo = DEG45_LEG_VALUE_OF_R3_EL_Servo;
        delay(25);

        L4_EL_Servo.write(DEG45_LEG_VALUE_OF_L4_EL_Servo);
        angleL4_EL_Servo = DEG45_LEG_VALUE_OF_L4_EL_Servo;
        delay(25);

        L3_EL_Servo.write(DEG45_LEG_VALUE_OF_L3_EL_Servo);
        angleL3_EL_Servo = DEG45_LEG_VALUE_OF_L3_EL_Servo;
        delay(25);

        R4_EL_Servo.write(DEG45_LEG_VALUE_OF_R4_EL_Servo);
        angleR4_EL_Servo = DEG45_LEG_VALUE_OF_R4_EL_Servo;
    }

    // -----------------------------------------------
    // SHOULDER WALKING STANCE
    // -----------------------------------------------

    else if (mode == "shoulder_walk") {

        R1_SHOUL_Servo.write(WALK_POS_R1_SHOULDER);
        angleR1_SHOUL_Servo = WALK_POS_R1_SHOULDER;
        delay(25);

        R2_SHOUL_Servo.write(WALK_POS_R2_SHOULDER);
        angleR2_SHOUL_Servo = WALK_POS_R2_SHOULDER;
        delay(25);

        L1_SHOUL_Servo.write(WALK_POS_L1_SHOULDER);
        angleL1_SHOUL_Servo = WALK_POS_L1_SHOULDER;
        delay(25);

        L2_SHOUL_Servo.write(WALK_POS_L2_SHOULDER);
        angleL2_SHOUL_Servo = WALK_POS_L2_SHOULDER;
    }

    else {

        giveAllServoMutexes();

        server.send(400, "text/plain", "Unknown preset mode");

        return;
    }

    giveAllServoMutexes();

    server.send(200, "text/plain", "Preset applied");
}

// =====================================================
// LEFT LEG TASK - CORE 0
// =====================================================

void leftLegTask(void *pvParameters) {

    Serial.println("Left leg task running on Core 0.");

    for (;;) {

        uint32_t currentPhase = 0;

        if (xTaskNotifyWait(
                0,
                0xFFFFFFFF,
                &currentPhase,
                pdMS_TO_TICKS(15)) == pdTRUE) {

            if (walking &&
                leftServoMutex != NULL &&
                xSemaphoreTake(
                    leftServoMutex,
                    pdMS_TO_TICKS(15)) == pdTRUE) {

                // Check again after acquiring the mutex.
                if (walking) {

                    if (currentPhase == 0) {

                        // Phase 1
                        int l4Val = 15;

                        L4_EL_Servo.write(l4Val);
                        angleL4_EL_Servo = l4Val;

                        L3_EL_Servo.write(WALK_BACK_L3_MAX_ANGLE);
                        angleL3_EL_Servo = WALK_BACK_L3_MAX_ANGLE;
                    }

                    else {

                        // Phase 2
                        int l4Val = WALK_FRONT_L4_MAX_ANGLE;

                        L4_EL_Servo.write(l4Val);
                        angleL4_EL_Servo = l4Val;

                        int l3Val = constrain(
                            VERTICAL_LEG_VALUE_OF_L3_EL_Servo
                                - WALK_STEP + ADD,
                            15,
                            165
                        );

                        L3_EL_Servo.write(l3Val);
                        angleL3_EL_Servo = l3Val;
                    }
                }

                xSemaphoreGive(leftServoMutex);
            }
        }

        // Say Hi runs on Core 0.
        if (wavingHi &&
            leftServoMutex != NULL &&
            xSemaphoreTake(
                leftServoMutex,
                pdMS_TO_TICKS(10)) == pdTRUE) {

            if (wavingHi) {
                updateHi();
            }

            xSemaphoreGive(leftServoMutex);
        }

        // 360 Spin runs on Core 0.
        if (spinning360 &&
            leftServoMutex != NULL &&
            xSemaphoreTake(
                leftServoMutex,
                pdMS_TO_TICKS(10)) == pdTRUE) {

            if (spinning360) {
                update360();
            }

            xSemaphoreGive(leftServoMutex);
        }
    }
}

// =====================================================
// RIGHT LEG TASK - CORE 1
// =====================================================

void rightLegTask(void *pvParameters) {

    Serial.println("Right leg task running on Core 1.");

    for (;;) {

        if (walking) {

            int currentPhase = walkPhase;

            // Synchronize the left leg task.
            if (leftLegTaskHandle != NULL) {

                xTaskNotify(
                    leftLegTaskHandle,
                    (uint32_t)currentPhase,
                    eSetValueWithOverwrite
                );
            }

            if (rightServoMutex != NULL &&
                xSemaphoreTake(
                    rightServoMutex,
                    pdMS_TO_TICKS(15)) == pdTRUE) {

                // Check again after acquiring the mutex.
                if (walking) {

                    if (currentPhase == 0) {

                        // Phase 1
                        int r4Val = constrain(
                            VERTICAL_LEG_VALUE_OF_R4_EL_Servo
                                - WALK_STEP + ADD,
                            15,
                            165
                        );

                        R4_EL_Servo.write(r4Val);
                        angleR4_EL_Servo = r4Val;

                        int r3Val = constrain(
                            VERTICAL_LEG_VALUE_OF_R3_EL_Servo
                                + WALK_STEP - ADD,
                            15,
                            165
                        );

                        R3_EL_Servo.write(r3Val);
                        angleR3_EL_Servo = r3Val;
                    }

                    else {

                        // Phase 2
                        int r4Val = constrain(
                            VERTICAL_LEG_VALUE_OF_R4_EL_Servo
                                + WALK_STEP - ADD,
                            15,
                            165
                        );

                        R4_EL_Servo.write(r4Val);
                        angleR4_EL_Servo = r4Val;

                        R3_EL_Servo.write(WALK_BACK_R3_MAX_ANGLE);
                        angleR3_EL_Servo = WALK_BACK_R3_MAX_ANGLE;
                    }
                }

                xSemaphoreGive(rightServoMutex);
            }

            walkPhase = (walkPhase == 0) ? 1 : 0;

            vTaskDelay(pdMS_TO_TICKS(DD));
        }

        else {

            vTaskDelay(pdMS_TO_TICKS(20));
        }
    }
}

// =====================================================
// WEB OTA LOCK STATE
// =====================================================

bool webOtaLocksHeld = false;

// =====================================================
// SETUP
// =====================================================

void setup() {

    // Retained from your original code.
    // A sound servo power supply is still required.
    WRITE_PERI_REG(RTC_CNTL_BROWN_OUT_REG, 0);

    Serial.begin(115200);

    delay(1000);

    Serial.println();
    Serial.println("==========================================");
    Serial.println("ESP32 ROBOT CONTROLLER & CALIBRATOR");
    Serial.println("==========================================");

    // -------------------------------------------------
    // 1. WIFI CONNECTION
    // -------------------------------------------------

    WiFi.mode(WIFI_STA);

    WiFi.setAutoReconnect(true);

    WiFi.setTxPower(WIFI_POWER_17dBm);

    WiFi.begin(ssid, password);

    Serial.print("Connecting to Wi-Fi");

    int attempts = 0;

    while (WiFi.status() != WL_CONNECTED && attempts < 40) {

        delay(500);

        Serial.print(".");

        attempts++;
    }

    Serial.println();

    if (WiFi.status() == WL_CONNECTED) {

        Serial.println("Wi-Fi connected.");

        Serial.print("ESP32 IP Address: ");

        Serial.println(WiFi.localIP());

        Serial.print("Signal strength: ");

        Serial.print(WiFi.RSSI());

        Serial.println(" dBm");
    }

    else {

        Serial.println("Wi-Fi connection failed.");
    }

    // -------------------------------------------------
    // 2. CREATE MUTEXES
    // -------------------------------------------------

    leftServoMutex = xSemaphoreCreateMutex();

    rightServoMutex = xSemaphoreCreateMutex();

    if (leftServoMutex == NULL || rightServoMutex == NULL) {

        Serial.println("ERROR: Servo mutex creation failed.");

        while (true) {
            delay(1000);
        }
    }

    // -------------------------------------------------
    // 3. INITIALIZE SERVO PWM
    // -------------------------------------------------

    ESP32PWM::allocateTimer(0);
    ESP32PWM::allocateTimer(1);
    ESP32PWM::allocateTimer(2);
    ESP32PWM::allocateTimer(3);

    R3_EL_Servo.setPeriodHertz(50);
    L4_EL_Servo.setPeriodHertz(50);
    L3_EL_Servo.setPeriodHertz(50);
    R4_EL_Servo.setPeriodHertz(50);

    R1_SHOUL_Servo.setPeriodHertz(50);
    R2_SHOUL_Servo.setPeriodHertz(50);
    L1_SHOUL_Servo.setPeriodHertz(50);
    L2_SHOUL_Servo.setPeriodHertz(50);

    // -------------------------------------------------
    // 4. ATTACH ELBOW SERVOS
    // -------------------------------------------------

    R3_EL_Servo.attach(R3_ELBOW);
    delay(25);

    L4_EL_Servo.attach(L4_ELBOW);
    delay(25);

    L3_EL_Servo.attach(L3_ELBOW);
    delay(25);

    R4_EL_Servo.attach(R4_ELBOW);
    delay(25);

    Serial.println("Elbow servos attached.");

    // -------------------------------------------------
    // 5. ATTACH SHOULDER SERVOS
    // -------------------------------------------------

    int chR1 = R1_SHOUL_Servo.attach(R1_SHOULDER);
    delay(25);

    int chR2 = R2_SHOUL_Servo.attach(R2_SHOULDER);
    delay(25);

    int chL1 = L1_SHOUL_Servo.attach(L1_SHOULDER);
    delay(25);

    int chL2 = L2_SHOUL_Servo.attach(L2_SHOULDER);
    delay(25);

    Serial.println("Shoulder servos attached.");

    Serial.printf(
        "R1 GPIO %d, channel %d\n",
        R1_SHOULDER,
        chR1
    );

    Serial.printf(
        "R2 GPIO %d, channel %d\n",
        R2_SHOULDER,
        chR2
    );

    Serial.printf(
        "L1 GPIO %d, channel %d\n",
        L1_SHOULDER,
        chL1
    );

    Serial.printf(
        "L2 GPIO %d, channel %d\n",
        L2_SHOULDER,
        chL2
    );

    // -------------------------------------------------
    // 6. INITIALIZE ALL EIGHT DEFAULT POSITIONS
    // -------------------------------------------------

    R3_EL_Servo.write(DEFAULT_R3_ELBOW);
    delay(25);

    L4_EL_Servo.write(DEFAULT_L4_ELBOW);
    delay(25);

    L3_EL_Servo.write(DEFAULT_L3_ELBOW);
    delay(25);

    R4_EL_Servo.write(DEFAULT_R4_ELBOW);
    delay(25);

    R1_SHOUL_Servo.write(DEFAULT_R1_SHOULDER);
    delay(25);

    R2_SHOUL_Servo.write(DEFAULT_R2_SHOULDER);
    delay(25);

    L1_SHOUL_Servo.write(DEFAULT_L1_SHOULDER);
    delay(25);

    L2_SHOUL_Servo.write(DEFAULT_L2_SHOULDER);

    Serial.println();
    Serial.println("Startup default positions:");
    Serial.printf("R1 Shoulder: %d\n", DEFAULT_R1_SHOULDER);
    Serial.printf("R2 Shoulder: %d\n", DEFAULT_R2_SHOULDER);
    Serial.printf("L1 Shoulder: %d\n", DEFAULT_L1_SHOULDER);
    Serial.printf("L2 Shoulder: %d\n", DEFAULT_L2_SHOULDER);

    Serial.printf("R3 Elbow: %d\n", DEFAULT_R3_ELBOW);
    Serial.printf("L4 Elbow: %d\n", DEFAULT_L4_ELBOW);
    Serial.printf("L3 Elbow: %d\n", DEFAULT_L3_ELBOW);
    Serial.printf("R4 Elbow: %d\n", DEFAULT_R4_ELBOW);

    // -------------------------------------------------
    // 7. START DUAL-CORE TASKS
    // -------------------------------------------------

    BaseType_t leftTaskResult = xTaskCreatePinnedToCore(
        leftLegTask,
        "LeftLegTask",
        4096,
        NULL,
        2,
        &leftLegTaskHandle,
        0
    );

    BaseType_t rightTaskResult = xTaskCreatePinnedToCore(
        rightLegTask,
        "RightLegTask",
        4096,
        NULL,
        2,
        &rightLegTaskHandle,
        1
    );

    if (leftTaskResult != pdPASS || rightTaskResult != pdPASS) {

        Serial.println("ERROR: Could not create walking tasks.");
    }

    // -------------------------------------------------
    // 8. HTTP ROUTES
    // -------------------------------------------------

    server.on("/", HTTP_GET, handleRoot);

    server.on("/servo", HTTP_GET, handleServo);

    server.on("/preset", HTTP_GET, handlePreset);

    server.on("/walk", HTTP_GET, handleWalk);

    server.on("/walkstatus", HTTP_GET, handleWalkStatus);

    server.on("/hi", HTTP_GET, handleHi);

    server.on("/starthi", HTTP_GET, []() {

        startHi();

        server.send(200, "text/plain", "Waving Hi (Continuous)");
    });

    server.on("/stophi", HTTP_GET, []() {

        stopHi();

        server.send(200, "text/plain", "Say Hi Stopped");
    });

    server.on("/360", HTTP_GET, handle360);

    server.on("/start360", HTTP_GET, []() {

        start360(SPIN360_DEFAULT_CYCLES);

        server.send(200, "text/plain", "360 Spin Started (23 Cycles)");
    });

    server.on("/stop360", HTTP_GET, []() {

        stop360();

        server.send(200, "text/plain", "360 Spin Stopped");
    });

    server.on("/ip", HTTP_GET, handleIP);

    server.on("/angles", HTTP_GET, handleAngles);

    // -------------------------------------------------
    // 9. BROWSER OTA ROUTES
    // -------------------------------------------------

    server.on("/update", HTTP_GET, []() {

        server.send_P(200, "text/html", OTA_page);
    });

    server.on(
        "/update",
        HTTP_POST,

        // Final response after upload processing.
        []() {

            server.sendHeader("Connection", "close");

            if (Update.hasError()) {

                server.send(
                    500,
                    "text/plain",
                    "Firmware update failed."
                );
            }

            else {

                server.send(
                    200,
                    "text/plain",
                    "Firmware update successful. Rebooting."
                );

                delay(500);

                ESP.restart();
            }
        },

        // Firmware data upload callback.
        []() {

            HTTPUpload& upload = server.upload();

            if (upload.status == UPLOAD_FILE_START) {

                Serial.printf(
                    "Browser OTA started: %s\n",
                    upload.filename.c_str()
                );

                walking = false;
                wavingHi = false;

                takeAllServoMutexes();

                webOtaLocksHeld = true;

                if (!Update.begin(UPDATE_SIZE_UNKNOWN)) {

                    Update.printError(Serial);
                }
            }

            else if (upload.status == UPLOAD_FILE_WRITE) {

                if (Update.write(
                    upload.buf,
                    upload.currentSize
                ) != upload.currentSize) {

                    Update.printError(Serial);
                }
            }

            else if (upload.status == UPLOAD_FILE_END) {

                if (Update.end(true)) {

                    Serial.printf(
                        "Browser OTA successful: %u bytes\n",
                        upload.totalSize
                    );
                }

                else {

                    Update.printError(Serial);
                }

                if (webOtaLocksHeld) {

                    giveAllServoMutexes();

                    webOtaLocksHeld = false;
                }
            }

            else if (upload.status == UPLOAD_FILE_ABORTED) {

                Serial.println("Browser OTA upload aborted.");

                Update.abort();

                if (webOtaLocksHeld) {

                    giveAllServoMutexes();

                    webOtaLocksHeld = false;
                }
            }
        }
    );

    server.begin();

    Serial.println("Web server started.");

    if (WiFi.status() == WL_CONNECTED) {

        Serial.print("Open the control page at: http://");

        Serial.println(WiFi.localIP());

        Serial.print("Browser OTA page: http://");

        Serial.print(WiFi.localIP());

        Serial.println("/update");
    }

    // -------------------------------------------------
    // 10. ARDUINO IDE OTA
    // -------------------------------------------------

    ArduinoOTA.setHostname("walking-robot");

    ArduinoOTA.onStart([]() {

        String type =
            (ArduinoOTA.getCommand() == U_FLASH)
                ? "sketch"
                : "filesystem";

        walking = false;
        wavingHi = false;

        takeAllServoMutexes();

        Serial.println(
            "ArduinoOTA started: " + type
        );
    });

    ArduinoOTA.onEnd([]() {

        Serial.println("ArduinoOTA update complete.");

        giveAllServoMutexes();
    });

    ArduinoOTA.onProgress([](
        unsigned int progress,
        unsigned int total
    ) {

        if (total > 0) {

            Serial.printf(
                "OTA Progress: %u%%\r",
                (progress * 100U) / total
            );
        }
    });

    ArduinoOTA.onError([](ota_error_t error) {

        Serial.printf(
            "ArduinoOTA Error[%u]\n",
            error
        );

        giveAllServoMutexes();
    });

    ArduinoOTA.begin();

    Serial.println("Arduino IDE OTA started.");
    Serial.println("OTA hostname: walking-robot.local");

    Serial.println("Dual-core robot controller initialized.");
}
